<script>

  let children = [{ age: 3 }];

  const MAX_CHILDREN = 3;

  const MAX_AGE = 17;

  

  function renderChildren() {

    const list = document.getElementById('children-list');

    list.innerHTML = '';

    children.forEach((child, i) => {

      const row = document.createElement('div');

      row.className = 'child-row fade-in';

      row.innerHTML = `

        <span class="child-label">第 ${i + 1} 位</span>

        <div class="age-input-wrap">

          <input type="number" class="age-input" min="0" max="${MAX_AGE}" value="${child.age}" data-idx="${i}">

          <span class="age-unit">歲</span>

          ${children.length > 1 ? `<button class="remove-btn" data-idx="${i}" title="移除">✕</button>` : ''}

        </div>

      `;

      list.appendChild(row);

    });

    

    document.querySelectorAll('.age-input').forEach(input => {

      input.addEventListener('input', e => {

        let v = parseInt(e.target.value);

        if (isNaN(v)) v = 0;

        if (v < 0) v = 0;

        if (v > MAX_AGE) v = MAX_AGE;

        children[parseInt(e.target.dataset.idx)].age = v;

        calculate();

      });

    });

    

    document.querySelectorAll('.remove-btn').forEach(btn => {

      btn.addEventListener('click', e => {

        children.splice(parseInt(e.target.dataset.idx), 1);

        renderChildren();

        calculate();

      });

    });

    

    document.getElementById('add-child-btn').disabled = children.length >= MAX_CHILDREN;

  }

  

  function calculate() {

    let monthlyCash = 0;

    let monthlyTisa = 0;

    let totalTisa = 0;

    let grandTotal = 0;

    const breakdown = [];

    

    children.forEach((child, i) => {

      const age = child.age;

      if (age < 0 || age > MAX_AGE) return;

      

      if (age < 6) {

        monthlyCash += 10000;

        breakdown.push(`第 ${i + 1} 位（${age} 歲）每月 10,000`);

      } else {

        monthlyCash += 2500;

        monthlyTisa += 2500;

        breakdown.push(`第 ${i + 1} 位（${age} 歲）每月 2,500 + TISA 2,500`);

      }

      

      const yearsUnder6 = Math.max(0, 6 - age);

      const yearsOver6 = Math.max(0, 18 - Math.max(age, 6));

      

      const cashUnder6 = yearsUnder6 * 12 * 10000;

      const cashOver6 = yearsOver6 * 12 * 2500;

      const tisaOver6 = yearsOver6 * 12 * 2500;

      

      totalTisa += tisaOver6;

      grandTotal += cashUnder6 + cashOver6 + tisaOver6;

    });

    

    document.getElementById('monthly-cash').textContent = monthlyCash.toLocaleString();

    document.getElementById('monthly-tisa').textContent = monthlyTisa.toLocaleString();

    document.getElementById('total-tisa').textContent = totalTisa.toLocaleString();

    document.getElementById('grand-total').textContent = grandTotal.toLocaleString();

    document.getElementById('cash-breakdown').textContent = 

      breakdown.length ? breakdown.join('　｜　') : '請輸入孩子年齡開始試算';

  }

  

  document.getElementById('add-child-btn').addEventListener('click', () => {

    if (children.length < MAX_CHILDREN) {

      children.push({ age: 8 });

      renderChildren();

      calculate();

    }

  });

  

  renderChildren();

  calculate();

</script>
